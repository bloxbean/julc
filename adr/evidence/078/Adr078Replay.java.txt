package replay;

import org.julclang.compiler.CompileResult;
import org.julclang.compiler.CompilerOptions;
import org.julclang.compiler.CompilerTargetRegistry;
import org.julclang.compiler.JulcCompiler;
import org.julclang.compiler.OptimizationLevel;
import org.julclang.core.PlutusData;
import org.julclang.core.flat.UplcFlatEncoder;
import org.julclang.core.text.UplcPrinter;
import org.julclang.stdlib.StdlibRegistry;
import org.julclang.vm.EvalResult;
import org.julclang.vm.ExBudget;
import org.julclang.vm.JulcVm;
import org.julclang.vm.OptimizationCostProfiles;

import java.math.BigInteger;
import java.nio.file.Files;
import java.nio.file.Path;
import java.security.MessageDigest;
import java.util.*;
import java.util.concurrent.*;

/**
 * ADR-078 M1a: ADR-075's evaluation differential (adr/evidence/075/EvalDiffReplayTest.java.txt) as a standalone runner,
 * so the same code runs on the pre21 artifacts and on the M1a build. On the ADR-074 corpus (20,407 recorded cases):
 * every compileMethod case with the default lookup is compiled at each recorded level and evaluated on ADR-075's
 * seven argument vectors under a transaction's limit (system property adr078.cpu/adr078.mem to raise it); every
 * validator case (compile, compileWithDetails) with the default lookup is compiled and its FLAT digest recorded.
 * Args: corpusRoot outFile [onlyKeysFile].
 */
public class Adr078Replay {
    static final ExBudget LIMIT = new ExBudget(Long.getLong("adr078.cpu", 10_000_000_000L), Long.getLong("adr078.mem", 14_000_000L));

    static final List<List<PlutusData>> VECTORS = List.of(
            List.of(),
            List.of(PlutusData.integer(BigInteger.ZERO)),
            List.of(PlutusData.integer(BigInteger.valueOf(7))),
            List.of(PlutusData.integer(BigInteger.valueOf(-3))),
            List.of(PlutusData.bytes(new byte[]{1, 2, 3})),
            List.of(PlutusData.integer(BigInteger.ONE), PlutusData.integer(BigInteger.TWO)),
            List.of(PlutusData.integer(BigInteger.valueOf(5)), PlutusData.bytes(new byte[]{9}), PlutusData.integer(BigInteger.TEN)));

    static final ThreadLocal<JulcVm> VM = ThreadLocal.withInitial(() -> {
        var vm = JulcVm.create("Java");
        var profile = OptimizationCostProfiles.PLUTUS_V3_PV11_COSTS_V1;
        vm.setCostModelParams(profile.costModelParameters(), profile.target());
        return vm;
    });

    public static void main(String[] a) throws Exception {
        Path root = Path.of(a[0]);
        Set<String> only = a.length > 2 ? new HashSet<>(Files.readAllLines(Path.of(a[2]))) : null;
        var cases = new ArrayList<Path>();
        try (var dirs = Files.list(root.resolve("cases"))) { dirs.sorted().forEach(cases::add); }
        var lines = new ConcurrentHashMap<String, String>();
        try (var pool = Executors.newFixedThreadPool(4)) {
            var futures = new ArrayList<Future<?>>();
            for (Path dir : cases) {
                if (only != null && !only.contains(dir.getFileName().toString())) continue;
                futures.add(pool.submit(() -> run(root, dir, lines)));
            }
            for (var f : futures) f.get();
        }
        var text = new StringBuilder();
        new TreeMap<>(lines).forEach((k, v) -> text.append(k).append(' ').append(v).append('\n'));
        Files.writeString(Path.of(a[1]), text.toString());
    }

    private static void run(Path root, Path dir, ConcurrentHashMap<String, String> lines) {
        String key = dir.getFileName().toString();
        try {
            var fields = new TreeMap<String, List<String>>();
            for (String line : Files.readString(dir.resolve("case.txt")).split("\n")) {
                int colon = line.indexOf(": ");
                if (colon > 0) fields.computeIfAbsent(line.substring(0, colon), k -> new ArrayList<>()).add(line.substring(colon + 2));
            }
            String entry = first(fields, "entry");
            String lookup = first(fields, "lookup");
            boolean method = "compileMethod".equals(entry);
            boolean validator = "compile".equals(entry) || "compileWithDetails".equals(entry);
            if (!(method || validator) || lookup == null || !lookup.startsWith("default:")) return;
            var libraries = new ArrayList<String>();
            for (String library : fields.getOrDefault("library", List.of())) libraries.add(source(root, library.split(" ")[0]));
            String main = source(root, first(fields, "source").split(" ")[0]);
            try (var files = Files.list(dir)) {
                for (String marker : files.map(p -> p.getFileName().toString()).filter(n -> n.startsWith("level-")).sorted().toList()) {
                    String level = marker.substring("level-".length());
                    var options = new CompilerOptions();
                    options.setTarget(CompilerTargetRegistry.targetForProfileId(first(fields, "target")));
                    options.setOptimizationLevel(OptimizationLevel.valueOf(level));
                    options.setSourceMapEnabled("true".equals(first(fields, "sourceMap")));
                    String cost = first(fields, "costProfile");
                    if (cost != null) options.setOptimizationCostProfile(OptimizationCostProfiles.forId(cost.split(" ")[0]));
                    String id = key + " " + level;
                    try {
                        var compiler = new JulcCompiler(StdlibRegistry.defaultRegistry(), options);
                        CompileResult result = method ? compiler.compileMethod(main, first(fields, "method"), libraries)
                                : compiler.compile(main, libraries);
                        if (result.hasErrors() || result.program() == null) { lines.put(id, "errors"); continue; }
                        lines.put(id + " flat", sha(UplcFlatEncoder.encodeProgram(result.program())));
                        if (!method) continue;
                        for (int v = 0; v < VECTORS.size(); v++) {
                            lines.put(id + " v" + v, describe(VM.get().evaluateWithArgs(result.program(), VECTORS.get(v), LIMIT)));
                        }
                    } catch (Throwable e) {
                        lines.put(id, "threw " + e.getClass().getSimpleName());
                    }
                }
            }
        } catch (Exception e) {
            lines.put(key, "unreadable " + e);
        }
    }

    private static String describe(EvalResult outcome) {
        return switch (outcome) {
            case EvalResult.Success s -> "ok cpu=" + s.consumed().cpuSteps() + " mem=" + s.consumed().memoryUnits() + " traces=" + s.traces()
                    + " result=" + s.resultTerm().getClass().getSimpleName() + ":" + sha(UplcPrinter.print(s.resultTerm()).getBytes());
            case EvalResult.Failure f -> "fail cpu=" + f.consumed().cpuSteps() + " mem=" + f.consumed().memoryUnits() + " traces=" + f.traces()
                    + " reason=" + sha(String.valueOf(f.error()).getBytes());
            case EvalResult.BudgetExhausted b -> "exhausted cpu=" + b.consumed().cpuSteps() + " mem=" + b.consumed().memoryUnits()
                    + " traces=" + b.traces();
        };
    }

    private static String sha(byte[] bytes) {
        try {
            return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(bytes));
        } catch (java.security.NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }

    private static String first(TreeMap<String, List<String>> fields, String name) {
        var values = fields.get(name);
        return values == null ? null : values.getFirst();
    }

    private static String source(Path root, String digest) throws Exception {
        return Files.readString(root.resolve("sources").resolve(digest + ".java"));
    }
}
