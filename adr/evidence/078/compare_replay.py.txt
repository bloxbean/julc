"""ADR-078 M1a: compares two Adr078Replay outputs (JSON lines). Exits 1 on any unaccepted difference.

Every record is validated against its kind and outcome: required fields present, no unknown kinds, statuses or
outcomes, no unknown fields. A harness error on either side fails the comparison, even where both sides match. Each
side must hold exactly one higher-limit retry for every evaluation that exhausted the transaction limit, and none
for any other (an orphan retry fails). The cases a run skipped must be exactly the pinned 55 (SKIPPED_SHA256).
Each side's records must be exactly the ones the corpus calls for (C29): one manifest record per case directory, with
the entry and lookup its case.txt records; one compile record per eligible case and level directory; seven
transaction-limit evaluations for each compileMethod case with the default lookup that compiled. Digests must be
64 lowercase hex digits.

Accepted, and reported apart: budget-only differences (I6b), of transaction-limit and of matched higher-limit
evaluations; bounded comparisons (both sides exhaust at both limits with the same traces); and, per ADR-078's I6a
rule, inconclusive cases (an exhaustion persists at the higher limit, and what it traced is a prefix of the other
side's traces). Where the transaction limit exhausted, the higher-limit retry is each side's evidence: where both
terminate, results and complete traces must be equal; traces that are not prefix-compatible are a mismatch (C36).
Vectors and budgets must be JSON integers, not booleans (C40). Where the higher limit resolves a one-sided exhaustion to
the same result, the two completed evaluations' budgets are reported as budget-only (C34). Everything else that
differs is unaccepted.
Usage: compare_replay.py corpus a.jsonl b.jsonl
"""
import hashlib, json, os, re, sys
from collections import Counter

SKIPPED_SHA256 = "b95d72fa4d276c2e5ac8875804dc500a869734430241f9347c39fd280dc82798"
SEMANTIC = ("outcome", "result", "reason", "traces")
COMMON = {"manifest": {"case", "kind", "status"}, "compile": {"case", "level", "kind", "status"},
          "eval": {"case", "level", "kind", "vector", "limit", "outcome"}}
REQUIRED = {
    ("manifest", "eligible"): {"entry", "lookup"}, ("manifest", "skipped"): {"entry", "lookup"},
    ("manifest", "harness-error"): {"error"},
    ("compile", "ok"): {"flat"}, ("compile", "errors"): {"diagnostics"}, ("compile", "threw"): {"error"},
    ("compile", "harness-error"): {"error"},
    ("eval", "ok"): {"result", "traces", "cpu", "mem"}, ("eval", "fail"): {"reason", "traces", "cpu", "mem"},
    ("eval", "exhausted"): {"traces", "cpu", "mem"}, ("eval", "harness-error"): {"error"},
}

VECTORS = 7
HEX = re.compile(r"[0-9a-f]{64}")
THREW = re.compile(r"[A-Za-z_$][\w.$]* [0-9a-f]{64}")
DIGESTS = ("flat", "result", "reason", "diagnostics")

def fail(message):
    sys.exit(message)

def validate(path, number, r):
    if not isinstance(r, dict): fail(f"{path}:{number}: not a JSON object")
    kind = r.get("kind")
    if kind not in COMMON: fail(f"{path}:{number}: unknown kind {kind!r}")
    if not COMMON[kind] <= r.keys(): fail(f"{path}:{number}: {kind} record lacks {sorted(COMMON[kind] - r.keys())}")
    state = r["status"] if kind != "eval" else r["outcome"]
    required = REQUIRED.get((kind, state))
    if required is None: fail(f"{path}:{number}: unknown {kind} state {state!r}")
    if not required <= r.keys(): fail(f"{path}:{number}: {kind}/{state} record lacks {sorted(required - r.keys())}")
    extra = r.keys() - COMMON[kind] - required
    if extra: fail(f"{path}:{number}: {kind}/{state} record has unknown fields {sorted(extra)}")
    if kind == "eval":
        if r["limit"] not in ("transaction", "high"): fail(f"{path}:{number}: unknown limit {r['limit']!r}")
        if type(r["vector"]) is not int: fail(f"{path}:{number}: vector is not an integer")
        if "traces" in r and not (isinstance(r["traces"], list) and all(isinstance(t, str) for t in r["traces"])):
            fail(f"{path}:{number}: traces is not a list of strings")
        for field in ("cpu", "mem"):
            if field in r and type(r[field]) is not int: fail(f"{path}:{number}: {field} is not an integer")
    for field in DIGESTS:
        if field in r and not (isinstance(r[field], str) and HEX.fullmatch(r[field])):
            fail(f"{path}:{number}: {field} is not a SHA-256 digest: {r[field]!r}")
    if kind == "compile" and state == "threw" and not (isinstance(r["error"], str) and THREW.fullmatch(r["error"])):
        fail(f"{path}:{number}: a rejection is not a class name and a message digest: {r['error']!r}")

def corpus(root):
    """Per case: (entry, lookup prefix, skipped, levels), from case.txt and the level directories, as the runner reads them."""
    cases = {}
    for key in sorted(os.listdir(os.path.join(root, "cases"))):
        directory = os.path.join(root, "cases", key)
        fields = {}
        for line in open(os.path.join(directory, "case.txt"), encoding="utf-8").read().split("\n"):
            name, colon, value = line.partition(": ")
            if colon and name not in fields: fields[name] = value
        entry, lookup = fields["entry"], fields["lookup"]
        skipped = entry == "JavaLibraryProvider" or lookup.startswith("custom:")
        levels = sorted(n[len("level-"):] for n in os.listdir(directory) if n.startswith("level-"))
        cases[key] = (entry, lookup.split(":")[0], skipped, levels)
    return cases

def expected(path, records, cases):
    """Fails unless the records are exactly those the corpus calls for, given this side's compile outcomes."""
    want = set()
    for key, (entry, lookup, skipped, levels) in cases.items():
        want.add((key, "", "manifest", "", ""))
        r = records.get((key, "", "manifest", "", ""))
        if r is not None and (r["status"], r.get("entry"), r.get("lookup")) != ("skipped" if skipped else "eligible", entry, lookup):
            fail(f"{path}: manifest record of {key} is not the corpus's: {r}")
        if skipped: continue
        for level in levels:
            want.add((key, level, "compile", "", ""))
            c = records.get((key, level, "compile", "", ""))
            if c is not None and c["status"] == "ok" and entry == "compileMethod" and lookup == "default":
                want.update((key, level, "eval", v, "transaction") for v in range(VECTORS))
    have = {rid for rid in records if rid[4] != "high"}
    missing, extra = sorted(want - have), sorted(have - want)
    if missing: fail(f"{path}: {len(missing)} record(s) the corpus calls for are missing, e.g. {missing[0]}")
    if extra: fail(f"{path}: {len(extra)} record(s) the corpus does not call for, e.g. {extra[0]}")

def load(path):
    records = {}
    for number, line in enumerate(open(path, encoding="utf-8"), 1):
        try:
            r = json.loads(line)
        except json.JSONDecodeError as e:
            fail(f"{path}:{number}: malformed record: {e}")
        validate(path, number, r)
        rid = (r["case"], r.get("level", ""), r["kind"], r.get("vector", ""), r.get("limit", ""))
        if rid in records: fail(f"{path}:{number}: duplicate record {rid}")
        records[rid] = r
    # Exactly one higher-limit retry for each transaction-limit exhaustion, and none for anything else.
    for rid, r in records.items():
        if r["kind"] != "eval": continue
        if rid[4] == "transaction" and r["outcome"] == "exhausted" and rid[:4] + ("high",) not in records:
            fail(f"{path}: exhausted evaluation {rid} has no higher-limit retry")
        if rid[4] == "high":
            t = records.get(rid[:4] + ("transaction",))
            if t is None or t["outcome"] != "exhausted": fail(f"{path}: orphan higher-limit retry {rid}")
    errors = [rid for rid, r in records.items() if r.get("status") == "harness-error" or r.get("outcome") == "harness-error"]
    if errors: fail(f"{path}: {len(errors)} harness error(s), e.g. {errors[0]}")
    skipped = sorted(r["case"] for r in records.values() if r["kind"] == "manifest" and r["status"] == "skipped")
    digest = hashlib.sha256(("\n".join(skipped) + "\n").encode()).hexdigest()
    if digest != SKIPPED_SHA256: fail(f"{path}: the {len(skipped)} skipped cases are not the pinned 55 (sha256 {digest})")
    return records

def semantic(r):
    return {k: r.get(k) for k in SEMANTIC}

def budget(r):
    return (r.get("cpu"), r.get("mem"))

cases = corpus(sys.argv[1])
a, b = load(sys.argv[2]), load(sys.argv[3])
expected(sys.argv[2], a, cases)
expected(sys.argv[3], b, cases)
failures, budget_only, inconclusive = [], [], []
counts = Counter()
for rid in sorted(set(a) | set(b)):
    if rid[4] == "high":
        continue  # compared with their transaction-limit record
    if rid not in a or rid not in b:
        failures.append(("on one side only", rid))
        continue
    x, y = a[rid], b[rid]
    kind = rid[2]
    if kind == "manifest":
        counts["manifests " + x["status"]] += 1
        if x != y: failures.append(("manifest differs", rid, x, y))
    elif kind == "compile":
        counts["compilations " + x["status"]] += 1
        if x != y: failures.append(("compilation differs", rid, x, y))
    else:
        ex, ey = x["outcome"] == "exhausted", y["outcome"] == "exhausted"
        hid = rid[:4] + ("high",)
        if ex or ey:
            # Each side's evidence is its transaction-limit outcome, or its higher-limit retry where the transaction
            # limit exhausted; a transaction-limit exhaustion's partial traces are superseded by the retry (C36).
            hx = a[hid] if ex else x
            hy = b[hid] if ey else y
            fx, fy = hx["outcome"] == "exhausted", hy["outcome"] == "exhausted"
            if not fx and not fy:
                if semantic(hx) != semantic(hy):
                    failures.append(("evaluation differs at the higher limit", rid, hx, hy))
                else:
                    counts["evaluations demonstrated " + hx["outcome"] + " (after an exhaustion)"] += 1
                    # One side ran out of the transaction budget and the other did not: always a budget difference
                    # (C34); both did: the retries' budgets are compared.
                    if ex != ey or budget(hx) != budget(hy): budget_only.append((rid, budget(hx), budget(hy)))
            elif fx and fy and ex and ey and semantic(hx) == semantic(hy):
                # Bounded: both exhaust at both limits, with the same traces; the retries' budgets are reported.
                counts["evaluations bounded (both exhausted at both limits, same traces)"] += 1
                if budget(hx) != budget(hy): budget_only.append((hid, budget(hx), budget(hy)))
            else:
                # An exhaustion persists: inconclusive only if what it traced is a prefix of the other side's traces
                # (complete, or, both exhausted, either way round); otherwise the traces already disagree (C36).
                tx, ty = hx["traces"], hy["traces"]
                prefix = lambda p, q: q[:len(p)] == p
                consistent = (prefix(tx, ty) or prefix(ty, tx)) if fx and fy else prefix(tx, ty) if fx else prefix(ty, tx)
                if consistent:
                    counts["evaluations inconclusive"] += 1
                    inconclusive.append(rid)
                else:
                    failures.append(("traces disagree before an exhaustion", rid, hx, hy))
        else:
            counts["evaluations demonstrated " + x["outcome"]] += 1
            if semantic(x) != semantic(y): failures.append(("evaluation differs", rid, x, y))
            elif budget(x) != budget(y): budget_only.append((rid, budget(x), budget(y)))

print("records:", len(a), "and", len(b))
for k in sorted(counts): print(f"  {k}: {counts[k]}")
print("unaccepted differences:", len(failures))
for f in failures[:20]: print("  ", f)
print("budget-only differences (I6b):", len(budget_only))
for d in budget_only[:20]: print("  ", d)
print("inconclusive (an exhaustion that persists at the higher limit):", len(inconclusive))
sys.exit(1 if failures else 0)
